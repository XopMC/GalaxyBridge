cmake_minimum_required(VERSION 3.25)
project(GalaxyBridge VERSION 0.1.0 LANGUAGES NONE)
if(NOT APPLE OR NOT CMAKE_HOST_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "This preset builds macOS arm64 plus the Android companion; use an Apple Silicon Mac.")
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(GB_BUILD_JOBS "4" CACHE STRING "Maximum native/Swift build concurrency")
set(GB_JAVA_HOME "$ENV{JAVA_HOME}" CACHE PATH "JDK 21.0.3+ (empty: standard macOS Java discovery)")
set(GB_ANDROID_SDK_ROOT "$ENV{ANDROID_SDK_ROOT}" CACHE PATH "Android SDK 37, Build Tools 37.0.0")
set(GB_ANDROID_NDK_HOME "$ENV{ANDROID_NDK_HOME}" CACHE PATH "Android NDK r29")
set(GB_RUSTC "$ENV{RUSTC}" CACHE FILEPATH "Optional rustc with Apple/Android arm64 targets")
set(GB_RUSTDOC "$ENV{RUSTDOC}" CACHE FILEPATH "Optional matching rustdoc")
set(GB_CARGO "$ENV{CARGO}" CACHE FILEPATH "Optional cargo executable")
set(GB_ENV "GB_BUILD_JOBS=${GB_BUILD_JOBS}" "CMAKE_BUILD_PARALLEL_LEVEL=${GB_BUILD_JOBS}")
foreach(PAIR "JAVA_HOME|GB_JAVA_HOME" "ANDROID_SDK_ROOT|GB_ANDROID_SDK_ROOT" "ANDROID_NDK_HOME|GB_ANDROID_NDK_HOME" "RUSTC|GB_RUSTC" "RUSTDOC|GB_RUSTDOC" "CARGO|GB_CARGO")
  string(REPLACE "|" ";" FIELDS "${PAIR}")
  list(GET FIELDS 0 ENV_NAME)
  list(GET FIELDS 1 CACHE_NAME)
  if(NOT "${${CACHE_NAME}}" STREQUAL "")
    list(APPEND GB_ENV "${ENV_NAME}=${${CACHE_NAME}}")
  endif()
endforeach()
function(gb_target TARGET ACTION)
  add_custom_target(${TARGET}
    COMMAND "${CMAKE_COMMAND}" -E env ${GB_ENV} "${Python3_EXECUTABLE}"
      "${CMAKE_SOURCE_DIR}/scripts/public-build.py" "${ACTION}" --build-dir "${CMAKE_BINARY_DIR}"
    USES_TERMINAL VERBATIM
    COMMENT "Galaxy Bridge: ${ACTION}")
endfunction()
gb_target(native native)
gb_target(macos-app macos-app)
gb_target(android-apk android-apk)
gb_target(package package)
gb_target(check check)
add_dependencies(macos-app native)
add_dependencies(package macos-app android-apk)
add_dependencies(check native)
# check compiles and tests source artifacts; it never installs/starts an app or phone service.
